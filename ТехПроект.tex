\section{Технический проект}
\subsection{Функция шифрования}
\textbf{Имя функции:} encrypt

\textbf{Описание интерфейса:}
\begin{itemize}
	\item \textbf{Входные параметры:}
	\begin{itemize}
		\item {data} — блок исходных данных для шифрования (16 байт);
		\item {key} — секретный ключ для шифрования (16 байт).
	\end{itemize}
	\item \textbf{Возвращаемое значение:} зашифрованный блок данных (16 байт);
	\item \textbf{Назначение:} выполняет процесс шифрования входного блока данных по ключу через побитового сложения XOR.
\end{itemize}
